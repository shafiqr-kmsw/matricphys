  <script>
    let performanceChartInstance = null;

    // Fetch data on page load with default ID
    window.addEventListener('DOMContentLoaded', () => {
      fetchStudentData();
    });

    function fetchStudentData() {
      const studentId = document.getElementById('studentId').value.trim();
      if (!studentId) return;

      google.script.run
        .withSuccessHandler(renderDashboard)
        .withFailureHandler(err => alert("Error: " + err.message))
        .getStudentData(studentId);
    }

    function renderDashboard(records) {
      if (!records || records.length === 0) {
        alert("No records found for the given Student ID.");
        return;
      }

      // 1. Populate Profile Card
      document.getElementById('dispId').innerText = records[0].id || '--';
      document.getElementById('dispName').innerText = records[0].name || '--';
      document.getElementById('dispClass').innerText = records[0].class || '--';

      // 2. Populate Academic Evaluation Matrix Table
      const tbody = document.getElementById('matrixTableBody');
      tbody.innerHTML = '';

      let totalA1 = 0, countA1 = 0;
      let totalA2 = 0, countA2 = 0;
      let totalA3 = 0, countA3 = 0;
      let cgpaSum = 0;

      const chartDatasets = [];
      const colors = ['#2563eb', '#db2777', '#059669', '#d97706', '#9333ea'];

      records.forEach((rec, index) => {
        const a1Val = isNumeric(rec.a1) ? Number(rec.a1) : null;
        const a2Val = isNumeric(rec.a2) ? Number(rec.a2) : null;
        const a3Val = isNumeric(rec.a3) ? Number(rec.a3) : null;

        if (a1Val !== null) { totalA1 += a1Val; countA1++; }
        if (a2Val !== null) { totalA2 += a2Val; countA2++; }
        if (a3Val !== null) { totalA3 += a3Val; countA3++; }

        // Calculate Subject Mean % and CGPA (Scale: 100% = 4.0 CGPA)
        const validScores = [a1Val, a2Val, a3Val].filter(v => v !== null);
        const meanPct = validScores.length ? (validScores.reduce((a, b) => a + b, 0) / validScores.length) : null;
        const cgpa = meanPct !== null ? (meanPct * 0.04) : null;
        if (cgpa !== null) cgpaSum += cgpa;

        const row = document.createElement('tr');
        row.innerHTML = `
          <td class="td-subject">${rec.subject || '--'}</td>
          <td>${a1Val !== null ? a1Val : 'N/A'}</td>
          <td>${a2Val !== null ? a2Val : 'N/A'}</td>
          <td>${a3Val !== null ? a3Val : 'N/A'}</td>
          <td class="highlight-blue">${meanPct !== null ? meanPct.toFixed(1) + '%' : '—'}</td>
          <td class="highlight-blue">${cgpa !== null ? cgpa.toFixed(2) : '—'}</td>
          <td class="status-not-found">${rec.targetGrade || 'Not Found'}</td>
          <td class="status-not-found">${rec.studentTargetGrade || 'Not Found'}</td>
        `;
        tbody.appendChild(row);

        // Prepare dataset for Chart
        chartDatasets.push({
          label: rec.subject,
          data: [a1Val || 0, a2Val || 0, a3Val || 0],
          borderColor: colors[index % colors.length],
          backgroundColor: colors[index % colors.length],
          borderWidth: 2,
          tension: 0.1,
          pointRadius: 4
        });
      });

      // Overall GPA calculations
      const avgA1GPA = countA1 > 0 ? ((totalA1 / countA1) * 0.04).toFixed(2) : 'N/A';
      const avgA2GPA = countA2 > 0 ? ((totalA2 / countA2) * 0.04).toFixed(2) : 'N/A';
      const avgA3GPA = countA3 > 0 ? ((totalA3 / countA3) * 0.04).toFixed(2) : 'N/A';
      const overallCGPA = records.length > 0 ? (cgpaSum / records.length).toFixed(2) : '—';

      // Summary / Bottom Row
      const summaryRow = document.createElement('tr');
      summaryRow.className = 'summary-row';
      summaryRow.innerHTML = `
        <td class="td-subject">GRADE (GPA)</td>
        <td>${avgA1GPA}</td>
        <td>${avgA2GPA}</td>
        <td>${avgA3GPA}</td>
        <td>—</td>
        <td>${overallCGPA}</td>
        <td>N/A</td>
        <td>N/A</td>
      `;
      tbody.appendChild(summaryRow);

      // 3. Render Chart
      renderChart(chartDatasets);
    }

    function renderChart(datasets) {
      const ctx = document.getElementById('performanceChart').getContext('2d');
      
      if (performanceChartInstance) {
        performanceChartInstance.destroy();
      }

      performanceChartInstance = new Chart(ctx, {
        type: 'line',
        data: {
          labels: ['Assessment 1', 'Assessment 2', 'Assessment 3'],
          datasets: datasets
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          scales: {
            y: {
              min: 0,
              max: 100,
              title: {
                display: true,
                text: 'Scores (%)',
                font: { weight: 'bold' }
              },
              ticks: { stepSize: 10 }
            }
          },
          plugins: {
            legend: {
              position: 'top',
              labels: { usePointStyle: true, boxWidth: 8 }
            }
          }
        }
      });
    }

    function isNumeric(val) {
      return !isNaN(parseFloat(val)) && isFinite(val);
    }
